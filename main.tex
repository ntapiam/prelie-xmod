\section{Preliminaries}
\subsection{pre-Lie algebras}
In this section we recall some definitions and facts about pre-Lie algebras that we will use in the following sections.
\begin{definition}
  A (left) pre-Lie algebra is a pair \((L,\prelie)\) with \(L\) a vector space and \(\prelie\colon L\otimes L\to L\) a bilinear operation such that the associator
  \begin{equation}\label{eq:PL}
    \mathrm{a}_{\prelie}(x,y,z)\coloneqq (x\prelie y)\prelie z - x\prelie(y\prelie z)\tag{PL}
  \end{equation}
  is left symmetric, i.e., such that \(\mathrm{a}_{\prelie}(x,y,z) = \mathrm{a}_{\prelie}(y,x,z)\) for all \(x,y,z\in L\).
\end{definition}

It can be show that for any pre-Lie algebra \((L,\prelie)\) the pair \(\mathfrak{g}_L\coloneqq(L,[-,-])\) where the bracket
\[
  [x,y]\coloneqq x\prelie y - y\prelie x
\]
is always a Lie algebra, called the \emph{subadjacent Lie algebra of \(L\)}.

We now recall the Oudom-Guin construction of the universal enveloping algebra of \(L\), identifying as the Hopf algebra \((S(L),\ast,\Delta)\) where \(\Delta\) is the standard coproduct on \(S(L)\) and the associative product \(\ast\) is defined below.
As usual, we will use Sweedler's notation for the coproduct:
\[
  \Delta A = A_{(1)}\otimes A_{(2)},\quad A\in S(L).
\]
\begin{definition}
  The pre-Lie operation is extended to \(\prelie\colon S(L)\otimes S(L)\to S(L)\) via \(1\prelie A \coloneqq 0, A\prelie 1 \coloneqq A\), and recursively
  \begin{align}
    xA\prelie B &\coloneqq x\prelie (A\prelie B) - (x\prelie A)\prelie B,\label{eq:OG1}\tag{OG1}\\
    A\prelie BC &\coloneqq (A_{(1)}\prelie B)(A_{(2)}\prelie C).\label{eq:OG2}\tag{OG2}
  \end{align}
\end{definition}

We observe that if \(x\in L\), the mapping \(\ell_{x\prelie}\colon S(L)\to S(L)\) is the unique derivation extending \(\prelie\) to \(S(L)\), that is,
\[
  \ell_{x\prelie}(y_1\dotsm y_n) = \sum_{j=1}^{n}y_1\dotsm\ell_{x\prelie}(y_j)\dotsm y_n.
\]

\begin{definition}
  On \(S(L)\) define the product
  \[
    A\ast B \coloneqq (A_{(1)}\prelie B)A_{(2)}.
  \]
\end{definition}

We note that from \cref{eq:OG2} it follows that
\[
  A\ast x_1\dotsm x_n = (A_{(1)}\prelie x_1)\dotsm(A_{(n)}\prelie x_n)A_{(n+1)}
\]
which gives a slightly more direct way of computing \(A\ast B\).

\begin{proposition}
  \label{prop:laction}
  Define \(\laction\colon L \otimes S(L) \to S(L)\) by
  \[
    x\laction A \coloneqq xA + x\prelie A.
  \]
  Then \(S(L)\) is a left pre-Lie module over \(L\).
\end{proposition}
\begin{proof}
  Let \(x,y\in L\) and \(A\in S(L)\).
  Then
  \begin{align*}
    x\laction(y\laction A) &= x\laction(yA + y\prelie A) \\
    &= xyA + (x\prelie y)A + y(x\prelie A) + x(y\prelie A) + x\prelie(y\prelie A).
  \end{align*}
  Hence
  \begin{align*}
    x\laction(y\laction A) - y\laction(x\laction A) &= (x\prelie y - y\prelie x)A + (x\prelie y - y\prelie x)\prelie A \\
    &= (x\prelie y - y\prelie x)\laction A,
  \end{align*}
  where we have used that \(x\prelie (y\prelie A) = xy\prelie A + (x\prelie y)\prelie A\) which follows immediately from \cref{eq:OG1}.
\end{proof}

\section{Crossed modules of pre-Lie algebras}
We take the point of view of \cite{LRW2024}.
In this framework, a pre-Lie crossed module is a pre-Lie structure on a two term chain complex \((\partial\colon T_1\to T_0)\).
This means we have a chain map \(\prelie\colon(T_1\to T_0)\otimes(T_1\to T_0)\to(T_1\to T_0)\) whose associator satisfies \cref{eq:PL}
and the following rectangle and all its inner squares commute:
\[
  \begin{tikzcd}
  T_1\otimes T_1\ar[r,"\partial^\otimes"]\ar[d,"\prelie"]&T_1\otimes T_0\oplus T_0\otimes T_1\ar[d,"\prelie"]\ar[r,"\partial^\otimes"]&T_0\otimes T_0\ar[d,"\prelie"]\\
  0\ar[r]&T_1\ar[r,"\partial"]&T_0
  \end{tikzcd}
\]
where \(\partial^\otimes(x\otimes y) = \partial(x)\otimes y+(-1)^{|x||y|}x\otimes\partial(y)\).

Unpacking this definition by restricting to each degree we see that, equivalently, it corresponds to a pre-Lie algebra \(\prelie\colon T_0\otimes T_0\to T_0\) together with a left pre-Lie module structure \(\laction\colon T_0\otimes T_1\to T_1\), a right action \(\raction\colon T_1\otimes T_0\to T_1\) and a map \(\partial\colon T_1\to T_0\) satsifying, for all \(s,t\in T_0\) and \(\mathscr{s},\mathscr{t}\in T_1\):
\begin{align}
  (s\prelie t-t\prelie s)\laction\mathscr{t} &= s\laction(t\laction\mathscr{t}) - t\laction(s\laction\mathscr{t})\tag{M1}\label{eq:m1}\\
  (s\laction \mathscr{t})\raction t - s\laction(\mathscr{t}\raction t) &= (\mathscr{t}\raction s)\raction t - \mathscr{t}\raction(s\prelie t)\tag{M2}\label{eq:m2},
\end{align}
and the equivariance and Peiffer relations
\begin{align}
  \partial(s\laction\mathscr{s}+\mathscr{t}\raction t) &= s\prelie\partial(\mathscr{s}) + \partial(\mathscr{t})\prelie t\tag{EQUI}\label{eq:equi}\\
  \partial(\mathscr{s})\laction\mathscr{t} &= \mathscr{s}\raction\partial(\mathscr{t})\tag{PEIF}\label{eq:peif}.
\end{align}

\begin{proposition}
Define the map \(\preliealt\colon T_1\otimes T_1\to T_1\) by
\[
  \mathscr{s}\preliealt\mathscr{t}\coloneqq\partial(\mathscr{s})\laction\mathscr{t}.
\]
Then \(T_1\) is a pre-Lie algebra and \(\partial\colon T_1\to T_0\) is a pre-Lie morphism.
\end{proposition}

We stress the fact that this second pre-Lie product is not immediately defined by the pre-Lie structure on the complex \((\partial\colon T_1\to T_0)\) but has to be introduced a posteriori.

\begin{proof}
  To see that \(T_1\) is a pre-Lie algebra it suffices to note that by \cref{eq:equi} the associator
  \begin{align*}
    \mathrm{a}_{\preliealt}(\mathscr{r},\mathscr{s},\mathscr{t}) &=\partial(\partial(\mathscr{r})\laction\mathscr{s})\laction\mathscr{t}-\partial(\mathscr{r})\laction(\partial(\mathscr{s})\laction\mathscr{t})\\
                                                                 &= (\partial(\mathscr{r})\prelie\partial(\mathscr{s}))\laction\mathscr{t}-\partial(\mathscr{r})\laction(\partial(\mathscr{s})\laction\mathscr{t})
  \end{align*}
  and thus it is left symmetric by \cref{eq:m1}.
  The second statement follows immediately from \cref{eq:equi}.
\end{proof}

\begin{remark}
  This notion seems to be a 2-term \(\text{pre-Lie}_\infty\) algebra as introduced in \cite{CL2001}.
\end{remark}
\subsection{Pre-Lie crossed modules as internal categories}
Consider now the category \(\catpl\) of pre-Lie algebras with pre-Lie morphisms.
An internal category in \(\catpl\) is a pair \((C_0, C_1)\) of a pre-Lie algebra \((C_0,\prelie)\) of ``objects'' and a pre-Lie algebra \((C_1,\preliealt)\) of ``arrows'', maps \(s,t\colon C_1\to C_0\), the source and the target, an identity map \(i\colon C_0\to C_1\) and a composition \(\circ\colon C_1\times_{s,t} C_1\to C_1\) satisfying the standard axioms of a category, with all maps being pre-Lie morphisms.
The composition however satisfies the so-called \emph{interchange law}: whenever \(f,g,h,k\in C_1\) are composable
\[
  (f\circ g)\preliealt(h\circ k) = (f\preliealt h)\circ(g\preliealt k).
\]


\subsection{Free pre-Lie crossed modules}
We begin by constructing the free pre-Lie module over a vector space.
Given a pre-Lie algebra \((L, \prelie)\) and a vector space \(V\), set
\[
  \mathcal{M}_L(V)\coloneqq S(L)\otimes V
\]
with action
\[
  s\laction(A\otimes v)\coloneqq (s\laction A)\otimes v = sA\otimes v + (s\prelie A)\otimes v
\]
for all \(s\in L\), \(A\in S(L)\) and \(v\in V\).
From \cref{prop:laction} we readily obtain the following result.
\begin{proposition}\label{prop:MLV.left}
  \((\mathcal{M}_L(V),\laction)\) is a pre-Lie module over \(L\).
\end{proposition}
\begin{theorem}
  \((\mathcal{M}_L(V),\laction)\) is the free pre-Lie module over \(V\).
\end{theorem}
\begin{proof}
  Let \((N,\circ_{\mathrm{L}})\) be another pre-Lie module over \(L\) and \(\varphi\in \operatorname{Hom}_{\mathsf{Vect}}(V,N)\).
  Define the extension \(\hat{\varphi}\colon \mathcal{M}_L(V)\to N\) recursively by
  \begin{align*}
    \hat{\varphi}(1\otimes v) &\coloneqq \varphi(v)\\
    \hat{\varphi}(tA\otimes v) &\coloneqq t\circ_{\mathrm{L}}\hat{\varphi}(A\otimes v) - \hat{\varphi}(t\prelie A\otimes v).
  \end{align*}
  for \(t\in L, A\in S(L)\).

  If \(t\in L\) and \(A\in S(L)\) then
  \begin{align*}
    \hat{\varphi}(t\laction (A\otimes v)) &= \hat{\varphi}\left( tA\otimes v + t\prelie A\otimes v \right)\\
                                          &= t\circ_{\mathrm{L}}\hat{\varphi}(A\otimes v) - \hat{\varphi}(t\prelie A\otimes v) + \hat{\varphi}(t\prelie A\otimes v) \\
                                          &= t\circ_{\mathrm{L}}\hat{\varphi}(A\otimes v).
  \end{align*}
  so \(\hat{\varphi}\) is a morphism of pre-Lie modules.
\end{proof}

\begin{lemma}
	\label{lem:AHcommute}
	\REF{ \cite{brown1976g} }

	$G$ a category object in $\GRP$.
	%\begin{align*}
	%	A \coloneqq \ker \tDC, \quad H \coloneqq \ker \sDC.
	%\end{align*}
	% Then $A$ and $H$ commute.
	Then $\ker \tDC$ and $\ker \sDC$ commute.

\end{lemma}
\begin{proof}

	First, $\iDC(\groupUnit_{G_0}) = \groupUnit_{G_1}$
	since $\iDC$ is a group morphism.
	For $a \in \ker \tDC, h \in \sDC$ we then have
	\begin{align*}
		a \COMPDC h
		&=
		(\groupUnit_{G_1} a) \COMPDC (h \groupUnit_{G_1}) \\
		&=
		(\groupUnit_{G_1} \COMPDC h) (a \COMPDC \groupUnit_{G_1}) \\
		&=
		h a \\
		a \COMPDC h
		&=
		(a \groupUnit_{G_1}) \COMPDC (\groupUnit_{G_1} h) \\
		&=
		(a \COMPDC \groupUnit_{G_1}) (\groupUnit_{G_1} \COMPDC h) \\
		&=
		a h.
	\end{align*}
	
\end{proof}


\begin{definition}
	$F: \mathcal C \to \mathcal D$ is an \textbf{equivalence of categories} if there is a functor $G: \mathcal D \to \mathcal C$ and natural isomorphisms
	\begin{align*}
		\epsilon: G\COMP F \Rightarrow \id_{\mathcal D}, \qquad \eta: \id_{\mathcal C} \Rightarrow F\COMP G.
	\end{align*}
\end{definition}

\begin{theorem}
	\label{thm:double_group_crossed_module}
	\REF{ \cite{forrester2002group}, but it is missing the proof of the Peiffer identity. }
	Given 
	a crossed module of groups
	$(\feedbackGG: H \to G, \actionGG)$,
	the following
	\footnote{Note the, maybe surprising, order $h'h$
	in the definition of $\COMPDC$.
	This would look ``correct'' if we had used 
	the usual composition $b \circ a = a \COMP b$.}
	\begin{align*}
		G_1 &\coloneqq H \rtimes G \\
		G_0 &\coloneqq G \\
		\sDC( h,g ) &\coloneqq g \\
		\tDC( h,g ) &\coloneqq \feedbackGG(h) g \\
		\iDC( g ) &\coloneqq (\groupUnit_H, g) \\
		(h,g) \COMPDC (h',g') &\coloneqq (h' h, g),
	\end{align*}
	defines a category in $\GRP$.

	\medskip

	Given a category in $\GRP$
	\begin{center}
	% https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAHEB9ARhAF9S6TLnyEU3clVqMWbLsX5SYUAObwioAGYAnCAFskZEDghIJIOAAssmnEgC0AJmr1mrRMf6CQO-YeomZtRWNnaIztJubAjUDHQARjAMAArCeARs2lgqlnYCWroGiOaBiEZxiSlpomxYYNiwIC4y7iBYinxAA
	\begin{tikzcd}
		G_1 \arrow[r, "\tDC", shift left=2] \arrow[r, "\sDC"', shift right=2] & G_0 \arrow[l, "\iDC" description]
	\end{tikzcd}
	\end{center}
	the following defines a crossed module of groups
	\begin{align*}
		H                 &\coloneqq \ker \sDC \\
		G                 &\coloneqq G_0 \\
		\feedbackGG( h )  &\coloneqq \tDC(h) \\
		\actionGG_g( h )  &\coloneqq \iDC(g)\,h\,\iDC(g)^{-1}.
	\end{align*}

	The two constructions give an equivalence of categories, between
	the category of categories in $\GRP$
	(with internal functors as morphisms, \Cref{thm:category_internal})
	and the category of crossed modules of groups, with crossed morphisms as morphisms.
	% Then the double category of shells over $G$ is equivalent to the double category of shells over $H$.	
\end{theorem}
\begin{proof}
	First claim:
	we only verify that $\COMPDC$ is a group morphism
	\begin{align*}
		G_1 \times_{G_0} G_1 \to G_1,
	\end{align*}
	where
	\begin{align*}
		G_1 \times_{G_0} G_1 = \{ ( (h,g), (k, \feedbackGG(h) g) ) \in G_1^2 \}.
	\end{align*}
	Now
	\begin{align*}
		(h,g)   \COMPDC (k, \feedbackGG(h) g)    &= (k h, g) \\
		(h',g') \COMPDC (k', \feedbackGG(h') g') &= (k' h', g).
	\end{align*}
	So 
	\begin{align*}
		( (h,g)   \COMPDC (k, \feedbackGG(h) g) ) \bullet ( (h',g') \COMPDC (k', \feedbackGG(h') g') )
		&=
		(k h \actionGG_g( k' h' ), g g').
	\end{align*}
	On the other hand
	\begin{align*}
		( (h,g)\bullet (h',g') ) \COMPDC ( (k, \feedbackGG(h) g) \bullet (k', \feedbackGG(h') g') )
		&=
		(h \actionGG_g(h'), g g') \COMPDC (k \actionGG_{\feedbackGG(h) g}(k'), \feedbackGG(h) g \feedbackGG(h') g') \\
		&=
		(k \actionGG_{\feedbackGG(h) g}(k') h \actionGG_g(h'), g g') \\
		&=
		(k \actionGG_{\feedbackGG(h)}( \actionGG_g(k') ) h \actionGG_g(h'), g g') \\
		&=	
		(k h \actionGG_g(k') h^{-1} h \actionGG_g(h'), g g') \\
		&=
		(k h \actionGG_g(k') \actionGG_g(h'), g g') \\
		&=
		(k h \actionGG_g(k' h'), g g'),
	\end{align*}
	as desired.

	Second claim:
	we verify the Peiffer identity.
	Let $h, h' \in H = \ker \sDC$.
	Then $a \coloneqq \iDC(\feedbackGG(h))^{-1} h \in \ker \tDC$.
	By \Cref{lem:AHcommute} we hence have
	\begin{align*}
		\iDC\big(\feedbackGG(h)\big)^{-1}\ h\ h'
		=
		h'\ \iDC\big(\feedbackGG(h)\big)^{-1}\ h 
	\end{align*}
	i.e.
	\begin{align*}
		h\ h'\ h^{-1}
		=
		\iDC\big(\feedbackGG(h)\big)\ h'\ \iDC\big(\feedbackGG(h)\big)^{-1}
		=
		\actionGG_{\feedbackGG(h)}(h').
	\end{align*}
	% \url{https://photos.google.com/photo/AF1QipPp6OOgnmJomC4j1nNbQhc08PU8IpRvuAxy0CHn}

	We skip the details of the proof of the equivalence of categories.
\end{proof}

\begin{definition}
	\REF{\cite[Section 1.1]{loday1982spaces}}
	A \DEF{$\mathsf{cat}^1$-group}
	is a triple $(G,s,t)$ where $G$ is a group
	and $s,t$ a group endomorphism of $G$ such that
	\begin{enumerate}
		\item 
		$t \COMP s = t, \quad s \COMP t = s$,
	\item
		$[\ker s, \ker t] = 1$.
	\end{enumerate}
\end{definition}

\begin{theorem}

	Given a $\mathsf{cat}^1$-group,
	the following
	\begin{align*}
		G_1       &\coloneqq G \\
		G_0       &\coloneqq \im s \\
		\sDC( x ) &\coloneqq s(x) \\
		\tDC( x ) &\coloneqq t(x) \\
		\iDC( g ) &\coloneqq \text{ embedding of the subgroup $\im s$ } \\
		x \COMPDC x' &\coloneqq x' \iDC(\sDC(x'))^{-1} x,
	\end{align*}
	defines a category in $\GRP$.

\end{theorem}
\begin{proof}
	We note that $s$ is idemptotent:
	\begin{align}
		s \COMP s
		=
		s \COMP t \COMP s
		=
		s \COMP t
		=
		s.
		\label{eq:s_square}
	\end{align}

	\bigskip

	First, $\tDC$ is well-defined, 
	since $t(G) = s(t(G)) \subset \im s$.

	Then for $y = s(x) \in \im s$,
	\begin{align*}
		\tDC(\iDC(y))
		&= 
		\tDC(\iDC(s(x)))
		=
		\tDC(s(x))
		=
		t(s(x)) = s(x) = y, \\
		%%%%%
		\sDC(\iDC(y))
		&=
		\sDC(\iDC(s(x)))
		=
		\sDC(s(x))
		=
		s(s(x))
		=
		s(x) = y,
	\end{align*}
	where we used \eqref{eq:s_square}.

	\textbf{Properties of $\COMPDC$}

	If $\tDC(x) = \sDC(x')$,
	\begin{align*}
		\tDC( x \COMPDC x' )
		&=
		\tDC( x' \iDC(\sDC(x'))^{-1} x ) \\
		% =
		% \tDC( x' ) \tDC( \iDC(\sDC(x'))^{-1} ) \tDC( x ) \\
		&=
		\tDC( x' ) \sDC(x')^{-1} ) \tDC( x )
		=
		\tDC( x' ) \\
		%%%%%%%%%
		\sDC( x \COMPDC x' )
		&=
		\sDC( x' \iDC(\sDC(x'))^{-1} x ) \\
		&=
		\sDC( x' ) \sDC(x')^{-1} \sDC( x )
		=
		\sDC(x).
	\end{align*}
	Further,
	if also $\tDC(y) = \sDC(y')$,
	\begin{align*}
		(x \COMPDC x')  (y \COMPDC y')
		=
		x' \iDC(\sDC(x'))^{-1} x y' \iDC(\sDC(y'))^{-1} y.
	\end{align*}
	Since
	\begin{align*}
		\tDC( \iDC(\sDC(x'))^{-1} x )
		&=
		\sDC(x')^{-1} \tDC(x)
		=
		1 \\
		\sDC( y' \iDC(\sDC(y'))^{-1} )
		&=
		\sDC(y') \sDC(y')^{-1} = 1.
	\end{align*}
	Hence, by assumption, the two elements commute and we continue
	\begin{align*}
		(x \COMPDC x')  (y \COMPDC y')
		=
		x' y' \iDC(\sDC(y'))^{-1} \iDC(\sDC(x'))^{-1} x y.
		=
		x' y' \iDC(\sDC(x'y'))^{-1} x y.
		=
		(x y) \COMPDC (x' y').
	\end{align*}
	
\end{proof}
